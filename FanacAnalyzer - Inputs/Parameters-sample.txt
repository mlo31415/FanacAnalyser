# ======================================================================================================================
# SAMPLE parameters file for FanacAnalyzer.
#
# The live file is "FanacAnalyzer Parameters.txt", which sits in the same directory as the executable and is deliberately NOT in git
# (it holds paths particular to one machine).  This sample IS in git, so that the set of settings and their meanings
# survive a fresh clone.  To use it, copy it to "FanacAnalyzer Parameters.txt" beside the executable and adjust.
#
# Lines beginning with "#" are ignored, as is anything from a " #" to the end of a line.
# Settings are name=value.  Names are matched case-insensitively.
# A setting which is absent takes the default noted below -- it is not an error unless marked "required".
# ======================================================================================================================


# Root Directory is a path relative to the executable's directory, and everything else below is relative to it.
# (It can be just a "." in which case it's the same directory as the executable.)
# A directory named on the command line is overridden by this setting.       [default: the command line, else "."]
Root Directory=.

# The read-only inputs -- the control files, the report header/footer templates, the APA templates and the
# <apa>-bumpf.txt files -- are all looked for in Input Directory, so that several of these site-maintenance
# programs can share one working directory without their control files colliding.  A file which is not there is
# looked for in Root Directory itself, so an older flat layout still works.  ("People Canonical Names.txt" is the
# exception: FancyAnalyzer produces it, and it is read from Report Directory.)
#                                                                       [default: "FanacAnalyzer - Inputs"]
Input Directory=FanacAnalyzer - Inputs

# The root on fanac.org to analyze.  Normally left blank, which is the same as setting it to "fanzines".   [default: ""]
Website URL=fanac.org/fanzines

# Reports are written to Report Directory, taken to be a subdirectory of Root Directory.
# The name carries the program's own name because several of these site-maintenance programs are often
# run from one directory.                                        [default: "FanacAnalyzer - Reports"]
Report Directory=FanacAnalyzer - Reports

# Where the list of every issue carrying an APA mailing is written.  Nothing reads it any more -- the APA mailings
# report below now works directly from memory -- but it is a useful artifact.      [default: "mailings.csv"]
mailings csv file=FanacAnalyzer - Reports/mailings.csv

# If this is present and non-empty, reload "Saved Fanzine List.json" instead of re-reading all ~2000 index pages from
# fanac.org, and write that file after a real read.  Uncomment it to make repeated runs fast while working on a report.
# Remember that the saved list goes stale.                                          [default: absent, i.e. read the website]
#Use Saved Fanzine List=True


# ----------------------------------------------------------------------------------------------------
# The APA mailings report.  (This was the separate FanacMailings program until 2026-09-17.)
# Its pages are written to an APAs subdirectory of Report Directory; the three templates, the <apa>-bumpf.txt files
# and "APA Mailings.xlsx" are all looked for in Input Directory.
#
# The three templates are REQUIRED: if a setting or its file is missing, the APA report is skipped (with a message
# in the error log naming what was missing) and the rest of the run finishes normally.
#
# "APA Mailings.xlsx" -- Joe's table of mailing dates and Official Editors -- is optional.  Without it the mailing
# pages are still generated, but with no date and no OE.  Joe reissues it from time to time, so replacing it is a
# normal thing to do; it is in git along with the <apa>-bumpf.txt files so that a fresh clone can build these pages.
Template-Mailing=Template - Mailing.html
Template-APA=Template - APA.html
Template-allAPAs=Template - All APAs.html

# The APAs to report on.  An issue is placed in a mailing when its Mailing column starts with one of these names
# followed by a space and the mailing number.  REQUIRED.
#
# Names are matched with spaces, punctuation and case ignored, so "Shadow FAPA", "Shadow-FAPA" and "ShadowFAPA"
# are all the same APA and only one of them needs to be listed.  A nickname spelled with different letters is not,
# so an entry may give alternate spellings after the canonical one, separated by "|":
#     "FWD|FIDO"     a Mailing column saying either "FWD 11" or "FIDO 11" belongs to the APA displayed as "FWD"
# The first spelling is canonical: it names the APA's directory and is what the reports display.  The same spellings
# are also used to find the APA's sheet in "APA Mailings.xlsx".
Known APAs="APA", "ANZAPA", "APA:NESFA", "APA45", "APA-F", "APA-L", "APA-Q", "APA-V", "CULT", "EOD", "FAPA", "FHAPA", "FLAP", "FWD|FIDO", "IPSO", "KAPA", "Minneapa", "N'APA", "OMPA", "PADS", "ROMPA", "SAPS", "SFPA", "Shadow FAPA", "Shadow SFPA", "Spinoff", "TAPS", "Turbo-APA", "VAPA", "WOOF"
